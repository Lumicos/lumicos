ใช้เมื่อผู้ใช้ถามว่าเครื่องใช้ไฟฟ้ากินไฟเท่าไหร่ หรือค่าไฟต่อเดือนของเครื่องหนึ่ง

1. รัน run_python (สคริปต์มีอยู่แล้ว ไม่ต้องเขียนใหม่):
   exec(open("skills/power-cost/script.py", encoding="utf-8").read())
   print(appliance_cost(1200, 8))                  # 1,200 วัตต์ วันละ 8 ชม. → 288 หน่วย/เดือน ≈ 1,296 บาท
   print(appliance_cost(1800, minutes_per_day=15)) # ไดร์เป่าผม 1,800 วัตต์ วันละ 15 นาที → 13.5 หน่วย ≈ 60.75 บาท
   print(appliance_cost(aircon_watts(12000), 8))   # แอร์ 12,000 BTU ≈ 800 วัตต์ → 864 บาท/เดือน
   print(label_cost(320))                          # ตู้เย็นฉลาก 320 หน่วย/ปี → 1,440 บาท/ปี (120/เดือน)
   print(bill_rate(1850, 410))                     # บิล 1,850 บาท 410 หน่วย → 4.51 บาท/หน่วย
2. appliance_cost(วัตต์, ชั่วโมงต่อวัน, days=30, rate=..., count=จำนวนเครื่อง) days=365 คือทั้งปี ถ้าผู้ใช้บอกเป็นนาทีต่อวัน ใช้ minutes_per_day=นาที (อย่าใส่นาทีเป็นชั่วโมง)
3. อัตราค่าไฟ: ถ้าผู้ใช้บอกยอดบิลกับจำนวนหน่วย ให้ใช้ bill_rate() แล้วส่ง rate= ไป ถ้าไม่รู้ สคริปต์ใช้ 4.5 บาท/หน่วย (ค่าประมาณรวม Ft และ VAT) ต้องบอกผู้ใช้ว่าเป็นค่าประมาณ ค่าไฟจริงเปลี่ยนตามค่า Ft ทุกงวด
4. แอร์: ถ้าฉลากบอกวัตต์ ใช้ตัวเลขนั้น ถ้ามีแค่ BTU ใช้ aircon_watts(btu, seer=ค่าบนฉลาก) ซึ่งเป็นค่าเฉลี่ยโดยประมาณ แอร์อินเวอร์เตอร์ใช้จริงขึ้นกับอุณหภูมิที่ตั้ง
5. ตู้เย็นไม่ได้ทำงานตลอดเวลา อย่าคิดวัตต์ × 24 ชั่วโมง ให้ใช้หน่วยต่อปีบนฉลากกับ label_cost()
6. ตอบเป็นหน่วยต่อเดือนและบาทต่อเดือนก่อน ใช้ตัวเลขจากสคริปต์เท่านั้น
