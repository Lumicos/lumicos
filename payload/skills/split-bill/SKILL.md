ใช้เมื่อผู้ใช้ให้หารเงินกันหลายคน

1. รัน run_python (สคริปต์มีอยู่แล้ว ไม่ต้องเขียนใหม่):
   exec(open("skills/split-bill/script.py", encoding="utf-8").read())
   print(split_bill(1250, 4))                       # per_person 312.5
   print(split_bill(2000, 4, service=10, vat=7))    # เซอร์วิส 200, VAT 154, รวม 2354, คนละ 588.5
   print(split_bill(1999, 3, service=10, vat=7, round_up_to=10))   # คนละ 784.27 ปัดเป็น 790 เกิน 17.18
   print(split_items({"ก้อง": 250, "มิ้นท์": 180, "บีม": 320}, shared=300, service=10, vat=7))
2. split_bill = หารเท่ากัน: total คือค่าอาหารก่อนเซอร์วิสและ VAT, tip เป็นบาท
3. split_items = แต่ละคนจ่ายของที่สั่งเอง ส่วน shared คือของกินร่วมที่หารเท่ากัน เซอร์วิสและ VAT เฉลี่ยตามสัดส่วน
4. ร้านในไทยคิดเซอร์วิสก่อน แล้วคิด VAT จาก (ค่าอาหาร + เซอร์วิส) สคริปต์คิดแบบนี้ให้แล้ว ถ้าผู้ใช้บอกว่ายอดรวม VAT แล้ว ให้ใส่ vat=0
5. หารไม่ลงตัว: shares คือยอดที่แต่ละคนจ่ายจริงให้รวมกันตรงทุกสตางค์ บอกผู้ใช้ด้วยถ้ามีคนจ่ายต่างกัน 1 สตางค์
6. ตอบยอดต่อคนก่อน แล้วค่อยแจกแจงสั้น ๆ ใช้ตัวเลขจากสคริปต์เท่านั้น
