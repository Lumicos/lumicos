ใช้เมื่อผู้ใช้ถามเรื่องออมเงิน เงินฝาก หรือเป้าหมายเก็บเงิน

1. รัน run_python (สคริปต์มีอยู่แล้ว ไม่ต้องเขียนใหม่):
   exec(open("skills/savings-growth/script.py", encoding="utf-8").read())
   print(savings(5000, 10, rate=2))                   # ออมเดือนละ 5,000 นาน 10 ปี ดอกเบี้ย 2% → 663,598.30
   print(savings(0, 5, rate=1.5, initial=100000))     # ฝากก้อนเดียว 100,000 นาน 5 ปี → 107,783.37
   print(monthly_needed(1000000, 10, rate=2))         # อยากได้ 1 ล้านใน 10 ปี → เดือนละ 7,534.68
   print(months_to_goal(100000, 3000, rate=1.5))      # ออมเดือนละ 3,000 → 33 เดือน (2 ปี 9 เดือน)
2. rate คือดอกเบี้ยต่อปีเป็นเปอร์เซ็นต์: 2% ใส่ 2 (ไม่ใช่ 0.02) ไม่บอกดอกเบี้ย = 0
3. สคริปต์คิดแบบฝากทุกสิ้นเดือนและทบต้นรายเดือน ถ้าผู้ใช้บอกเงื่อนไขอื่น (ทบต้นรายปี ฝากต้นเดือน) ให้บอกว่าตัวเลขจะต่างเล็กน้อย
4. ภาษีดอกเบี้ย: ฝากประจำทั่วไปมักหัก ณ ที่จ่าย 15% ของดอกเบี้ย ใส่ tax=15 ส่วนออมทรัพย์มักได้รับยกเว้นถ้าดอกเบี้ยทั้งปีไม่เกิน 20,000 บาท ถ้าผู้ใช้ถามเงื่อนไขภาษีแบบละเอียด ให้แนะนำให้ตรวจกับธนาคาร
5. ดอกเบี้ยจริงของแต่ละธนาคารเปลี่ยนได้ ถ้าผู้ใช้ไม่บอกอัตรา ให้ถามหรือยกตัวอย่าง 2–3 อัตราให้เทียบ ห้ามอ้างว่าเป็นอัตราปัจจุบันของธนาคารใด
6. ตอบยอดสุดท้าย เงินที่ฝากเอง และดอกเบี้ยที่ได้ ใช้ตัวเลขจากสคริปต์เท่านั้น
