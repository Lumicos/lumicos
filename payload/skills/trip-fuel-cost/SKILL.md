ใช้เมื่อผู้ใช้ถามค่าน้ำมันหรือจำนวนลิตรของการเดินทางระยะหนึ่ง

1. รัน run_python (สคริปต์มีอยู่แล้ว ไม่ต้องเขียนใหม่):
   exec(open("skills/trip-fuel-cost/script.py", encoding="utf-8").read())
   print(trip_cost(450, 40, km_per_liter=12))                       # 37.5 ลิตร 1,500 บาท
   print(trip_cost(450, 40, km_per_liter=12, round_trip=True, people=3))  # ไป-กลับ 900 กม. หาร 3
   print(trip_cost(200, 38, liters_per_100km=7))                    # อัตราแบบ ลิตร/100 กม.
   print(ev_trip_cost(300, 7.5, kwh_per_100km=16))                  # รถไฟฟ้า 48 kWh 360 บาท
2. ใช้ราคาน้ำมันที่ผู้ใช้บอก ถ้าไม่บอกให้ถามราคาหรือบอกว่าคิดจากราคาไหน ห้ามเดาราคาเอง
3. ตอบจำนวนลิตรและค่าน้ำมันจากผลของสคริปต์ (มีลูกน้ำหลักพัน) ห้ามคิดเลขเอง
