1. รัน run_python (สคริปต์มีอยู่แล้ว ไม่ต้องเขียนใหม่):
   exec(open("skills/unit-price/script.py", encoding="utf-8").read())
   print(compare_prices(("ขวดใหญ่", 25, 2, "ลิตร"), ("แพ็ค 6", 72, 500, "ml", 6)))   # 12.5 กับ 24 บาท/ลิตร → ขวดใหญ่คุ้มกว่า
   print(per_unit(39, 700, "ml"))     # 55.71 บาท/ลิตร
